cmake_minimum_required(VERSION 3.16)
project(holylib_mcp_tests LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(holylib_mcp_tests native_tests.cpp)
target_compile_features(holylib_mcp_tests PRIVATE cxx_std_17)
target_include_directories(holylib_mcp_tests PRIVATE ../../source)
target_link_libraries(holylib_mcp_tests PRIVATE Threads::Threads)
if(WIN32)
  target_link_libraries(holylib_mcp_tests PRIVATE ws2_32 crypt32)
endif()
enable_testing()
add_test(NAME holylib_mcp_tests COMMAND holylib_mcp_tests)
set_tests_properties(holylib_mcp_tests PROPERTIES TIMEOUT 30)
